import java.util.*;

class ParkingLot {
    private final List<Floor> floors;
    private final FeePolicy fees;
    private final Map<String, Ticket> open = new HashMap<>();

    ParkingLot(List<Floor> floors, FeePolicy fees) {
        this.floors = floors;
        this.fees = fees;
    }

    Ticket park(Vehicle v, long nowMillis) {
        for (Floor f : floors) {
            Optional<Spot> spot = f.nearestFree(v.type());
            if (spot.isPresent()) {
                spot.get().occupy(v);
                Ticket t = new Ticket(v.plate(), spot.get(), nowMillis);
                open.put(t.id(), t);
                return t;
            }
        }
        throw new IllegalStateException("No spot for " + v.type());
    }
}
